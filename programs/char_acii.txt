; карта памяти:
; 3000-3010 под символы аски
; 4000-4010 под числа
; 3022 ячейка под готовое число
;
; 


label. #main
    
    assig. r4 '3000'
    assig. r5 '0'
    assig. r7 '0' регистр для счёт разрядов
    assig. r2 '0' регистр адреса аски
    assig. r8 '0' регистр адреса цифр дигитов
    assig. r2 '0'
    assig. r10 '3'

    global. "line_listen_cl.txt"
    assig. r7 '0'

    label. #digit_loop
        transf. r7 r2
        math. "i" r2 '3000'
        raml. "m" r1 r2
        go. #numb_loop
    label. #num_wrt
        transf. r7 r8
        math. "i" r8 '4000' 
        ramw. "m" r1 r8
        math. "i" r7 '1'
        if. r7 > r3 #end
        go. #digit_loop

    label. #numb_loop
            if r1 $ '48' #zero_label
            if r1 $ '49' #one_label
            if r1 $ '50' #two_label
            if r1 $ '51' #three_label
            if r1 $ '52' #four_label
            if r1 $ '53' #five_label
            if r1 $ '54' #six_label
            if r1 $ '55' #seven_label
            if r1 $ '56' #eigth_label
            if r1 $ '57' #nine_label

             label. #zero_label
                assig. r1 '0'
            go. #num_wrt
             label. #one_label
                assig. r1 '1'
            go. #num_wrt
             label. #two_label
                assig. r1 '2'
            go. #num_wrt
             label. #three_label
                assig. r1 '3'
            go. #num_wrt
             label. #four_label
                assig. r1 '4'
            go. #num_wrt
             label. #five_label
                assig. r1 '5'
            go. #num_wrt
             label. #six_label
                assig. r1 '6'
            go. #num_wrt
             label. #seven_label
                assig. r1 '7'
            go. #num_wrt
             label. #eigth_label
                assig. r1 '8'
            go. #num_wrt
               label. #nine_label
                assig. r1 '9'
            go. #num_wrt
            
    label. #end
    assig. r11 '1000'
    raml. "r" r9 &4000 
    math. r9 * r11 r9
    ramw. "r" r9 &4000

    assig. r11 '100'
    raml. "r" r9 &4001
    math. r9 * r11 r9
    ramw. "r" r9 &4001

    assig. r11 '10'
    raml. "r" r9 &4002
    math. r9 * r11 r9
    ramw. "r" r9 &4002

    assig. r11 '1'
    raml. "r" r9 &4003 
    math. r9 * r11 r9
    ramw. "r" r9 &4003
;   r 12 для пром. суммы
;   r11 для втор. слог
    
    raml. "r" r9 &4000
    raml. "r" r11 &4001
    math. r9 + r11 r12
    raml. "r" r11 &4002
    math. r12 + r11 r12
    raml. "r" r11 &4003
    math. r12 + r11 r12

    ramw. "r" r12 &3022
    raml. "r" r12 &3022
    out. "r" r12 P5


    halt.

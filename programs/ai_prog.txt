label. #main
    assig. r1 '0'      / Текущий X
    assig. r2 '0'      / Текущий Y
    assig. r3 '64'     / Ширина экрана
    assig. r4 '64'     / Высота экрана
label. #draw_line
    out. "r" r1 P1     / Передаем X в порт P1 (координата X)
    out. "r" r2 P2     / Передаем Y в порт P2 (координата Y)
    out. "n" '1' P3    / Рисуем пиксель (цвет 1) в порт P3
    math. "i" r1 '1'   / X = X + 1
    if. r1 < r3 #draw_line / Если не конец строки, продолжаем рисовать
label. #next_row
    assig. r1 '0'      / Сбрасываем X
    math. "i" r2 '1'   / Y = Y + 1
    / Если Y достиг высоты экрана, начинаем сначала (зацикливаем)
    if. r2 = r4 #break
    go. #draw_line     / Переходим к рисованию следующей строки
label. #break
global. 
halt.
